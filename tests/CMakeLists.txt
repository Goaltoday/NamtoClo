cmake_minimum_required(VERSION 3.24)
project(NamToCloFinalToneMatchTests LANGUAGES CXX)
enable_testing()
add_executable(final_tonematch_test final_tonematch_test.cpp)
target_compile_features(final_tonematch_test PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(final_tonematch_test PRIVATE /W4 /EHsc /fp:precise /utf-8)
else()
    target_compile_options(final_tonematch_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME final_tonematch COMMAND final_tonematch_test "${CMAKE_CURRENT_BINARY_DIR}/test-output")
add_test(NAME final_tonematch_long_tail COMMAND final_tonematch_test "${CMAKE_CURRENT_BINARY_DIR}/test-output-long" long-tail)
